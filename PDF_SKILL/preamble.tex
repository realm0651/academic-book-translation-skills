% Bidirectional endnote navigation. The Lua filter creates a unique target at
% every正文 note marker and appends these compact backlinks to the matching
% endnote entry. A repeated source note receives numbered return links.
% Robust正文 note marker; safe inside headings/bookmarks as well as normal text.
\DeclareRobustCommand{\booknoteref}[3]{%
  \hypertarget{#1}{}\hyperlink{#2}{\textsuperscript{#3}}%
}
\pdfstringdefDisableCommands{%
  \def\booknoteref#1#2#3{}%
}
\newcommand{\booknotebacklink}[1]{%
  \nobreak\hspace{0.38em}%
  \hyperlink{#1}{\raisebox{0.15ex}{\scriptsize\ensuremath{\hookleftarrow}}}%
}
\newcommand{\booknotebacklinkmulti}[2]{%
  \nobreak\hspace{0.32em}%
  \hyperlink{#1}{\raisebox{0.15ex}{\scriptsize\ensuremath{\hookleftarrow}\!#2}}%
}
