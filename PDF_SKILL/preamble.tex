% -----------------------------------------------------------------------------
% Display mathematics
% -----------------------------------------------------------------------------
% Measure at the normal display-math size and shrink ONLY when the natural box
% would exceed the text block. Unlike the old adjustbox pipeline, short equations
% are never enlarged to fill the line.
\newsavebox{\bookmathbox}
\newcommand{\bookfitmath}[1]{%
  \sbox{\bookmathbox}{$\displaystyle #1$}%
  \ifdim\wd\bookmathbox>.92\linewidth
    \resizebox{.92\linewidth}{!}{\usebox{\bookmathbox}}%
  \else
    \usebox{\bookmathbox}%
  \fi
}

% -----------------------------------------------------------------------------
% Figures
% -----------------------------------------------------------------------------
% Do NOT force every bitmap to \linewidth. Small source figures should remain at
% their natural size; large ones are reduced, never enlarged or cropped.
\newsavebox{\bookimagebox}
\newcommand{\bookincludegraphics}[1]{%
  \par\smallskip\noindent
  \begingroup
  \sbox{\bookimagebox}{\includegraphics{#1}}%
  \ifdim\wd\bookimagebox>.94\linewidth
    \sbox{\bookimagebox}{\resizebox{.94\linewidth}{!}{\usebox{\bookimagebox}}}%
  \fi
  \ifdim\ht\bookimagebox>.72\textheight
    \sbox{\bookimagebox}{\resizebox{!}{.72\textheight}{\usebox{\bookimagebox}}}%
  \fi
  \noindent\makebox[\linewidth][c]{\usebox{\bookimagebox}}%
  \endgroup
  \par\smallskip
}
\newsavebox{\booktableimagebox}
\newcommand{\bookincludetableimage}[1]{%
  \par\smallskip\noindent
  \begingroup
  \sbox{\booktableimagebox}{\includegraphics{#1}}%
  \ifdim\wd\booktableimagebox>.96\linewidth
    \sbox{\booktableimagebox}{\resizebox{.96\linewidth}{!}{\usebox{\booktableimagebox}}}%
  \fi
  \ifdim\ht\booktableimagebox>.76\textheight
    \sbox{\booktableimagebox}{\resizebox{!}{.76\textheight}{\usebox{\booktableimagebox}}}%
  \fi
  \noindent\makebox[\linewidth][c]{\usebox{\booktableimagebox}}%
  \endgroup
  \par\smallskip
}
